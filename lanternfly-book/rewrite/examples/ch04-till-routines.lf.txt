// Chapter 4 companion — Till routines
// The invoice program decomposed into named operations with private state.
// Verified against specification 0.4: §4.2, §8.7, §11.1–11.5.

var total as u16 = 0
var saleCount as u8 = 0

// Action: no result type. One transaction description, many transactions.
sub addSale(price as u16)
    total = total + price
    saleCount = saleCount + 1
end

// Value-producers: trailing `as Type`, every path returns.
sub poundsPart(amount as u16) as u16
    return amount / 100
end

sub pencePart(amount as u16) as u16
    return amount mod 100
end

// A local holds a private intermediate; fresh every invocation.
sub totalWithService(amount as u16) as u16
    var service as u16 = amount / 10

    return amount + service
end

// Precondition: saleCount > 0, or the division faults (F-DIV-ZERO).
// Chapter 5 turns this comment into a guard.
sub meanPrice() as u16
    return total / saleCount
end

var pounds as u16 = 0
var pence as u16 = 0
var mean as u16 = 0

sub main()
    addSale(120)
    addSale(95)

    total = totalWithService(total)   // 215 + 21 = 236
    pounds = poundsPart(total)        // 2
    pence = pencePart(total)          // 36
    mean = meanPrice()                // 236 / 2 = 118
end

// Trace:
//   addSale(120)          total = 120  saleCount = 1
//   addSale(95)           total = 215  saleCount = 2
//   totalWithService(215) service = 21, returns 236
//   poundsPart(236) = 2, pencePart(236) = 36, meanPrice() = 118
//   final storage: total 236, saleCount 2, pounds 2, pence 36, mean 118
//
// Spec notes:
// - Locals precede statements and initialize in order per invocation (§4.2).
// - `total / saleCount`: u8 widens to the u16 already present (§3.1).
// - Arguments evaluate left to right before invocation begins (§8.7).
// - Every reachable path in a result-bearing routine returns (§11.5) —
//   trivially true of these straight-line bodies; ch 5 makes it interesting.
