// ============================== model.lf ===============================
// Ch 10-11 behind ch 14's border: the table and its invariant, private.

export record Reading              // exported: report.lf renders fields
    var day as u8
    var month as u8
    var temperature as i16
    var validated as boolean
end

const logCapacity as u8 = 10

var log as Reading[logCapacity]    // private
var logCount as u8 = 0             // private

// Ch 4/13's rule met honestly: locals cannot own aggregates, so the
// sort's travelling value is one static scratch slot — the small-machine
// pattern the rule was always pointing at.
var carried as Reading             // private scratch for sortByTemperature

export sub addReading(day as u8, month as u8, temperature as i16) as boolean
    if logCount >= logCapacity then
        return false
    end

    log[logCount].day = day
    log[logCount].month = month
    log[logCount].temperature = temperature
    log[logCount].validated = true
    logCount = logCount + 1

    return true
end

export sub entryCount() as u8
    return logCount
end

export sub dayAt(index as u8) as u8
    return log[index].day
end

export sub temperatureAt(index as u8) as i16
    return log[index].temperature
end

// Ch 9's insertion sort, element type upgraded to Reading: the structure
// and the invariant transfer word for word; each shift is now a
// five-byte aggregate copy (§7), and the cost table gains that column.
export sub sortByTemperature()
    var pass as u8 = 0
    var slot as u8 = 0

    for pass = 1 until logCount
        carried = log[pass]
        slot = pass

        while slot > 0 and log[slot - 1].temperature > carried.temperature
            log[slot] = log[slot - 1]
            slot = slot - 1
        end

        log[slot] = carried
    end
end

export sub coldest() as i16
    var index as u8 = 0
    var lowest as i16 = 0

    if logCount = 0 then
        return 0
    end

    lowest = log[0].temperature

    for index = 1 until logCount
        if log[index].temperature < lowest then
            lowest = log[index].temperature
        end
    end

    return lowest
end

export sub warmest() as i16
    var index as u8 = 0
    var highest as i16 = 0

    if logCount = 0 then
        return 0
    end

    highest = log[0].temperature

    for index = 1 until logCount
        if log[index].temperature > highest then
            highest = log[index].temperature
        end
    end

    return highest
end

export sub meanTemperature() as i16
    var index as u8 = 0
    var sum as i16 = 0

    if logCount = 0 then
        return 0
    end

    for index = 0 until logCount
        sum = sum + log[index].temperature
    end

    return sum / logCount
end

// ============================= platform.lf =============================
// Ch 15's face, unchanged.

export extern sub printChar(ch as u8) at $0008
export extern sub printText(text as near cstr) from "ROM_PRINT_TEXT"
export extern sub printNumber(value as u16) from "ROM_PRINT_U16"
export extern sub newline() from "ROM_NEWLINE"

asm
ROM_PRINT_TEXT = $0028
ROM_PRINT_U16  = $002b
ROM_NEWLINE    = $002e
end

// ============================== report.lf ==============================
// Root module: composition and main. The build manifest names this file
// and main() (Q1: the manifest's format is the project's to fix).

import "model.lf"
import "platform.lf"

const title as near cstr = "WEEK REPORT"
const coldLabel as near cstr = "COLDEST "
const warmLabel as near cstr = "WARMEST "
const meanLabel as near cstr = "MEAN "

sub printSigned(value as i16)
    if value < 0 then
        printChar(u8('-'))
    end

    printNumber(abs(value))
end

sub printEntry(index as u8)
    printNumber(dayAt(index))
    printChar(u8(':'))
    printChar(u8(' '))
    printSigned(temperatureAt(index))
    newline()
end

sub main()
    var index as u8 = 0
    var seeded as boolean = false

    seeded = addReading(11, 1, 2)
    seeded = addReading(12, 1, -3)
    seeded = addReading(13, 1, 5)
    seeded = addReading(14, 1, 0)
    seeded = addReading(15, 1, -1)

    sortByTemperature()

    printText(title)
    newline()

    for index = 0 until entryCount()
        printEntry(index)
    end

    printText(coldLabel)
    printSigned(coldest())
    newline()

    printText(warmLabel)
    printSigned(warmest())
    newline()

    printText(meanLabel)
    printSigned(meanTemperature())
    newline()
end

// Sort trace (temperatures, | = sorted prefix):
//   start   [2 | -3, 5, 0, -1]
//   pass 1  [-3, 2 | 5, 0, -1]    carried -3, 1 copy
//   pass 2  [-3, 2, 5 | 0, -1]    carried 5,  0 copies
//   pass 3  [-3, 0, 2, 5 | -1]    carried 0,  2 copies
//   pass 4  [-3, -1, 0, 2, 5]     carried -1, 3 copies
//   6 element copies of 5 bytes each, plus 4 in and 4 out through carried.
// Stats: coldest -3, warmest 5, mean (2-3+5+0-1)/5 = 3/5 = 0.
//
// Ordered service trace (the program's observable, beside final storage):
//   printText("WEEK REPORT"), newline,
//   12: -3 / 15: -1 / 14: 0 / 11: 2 / 13: 5   (sorted day lines, each as
//     printNumber(day), ':', ' ', sign?, printNumber(magnitude), newline)
//   printText("COLDEST "), '-', printNumber(3), newline
//   printText("WARMEST "), printNumber(5), newline
//   printText("MEAN "), printNumber(0), newline
//
// Static budget, priced byte by byte (the book's thesis at the finish):
//   log 50 + logCount 1 + carried 5 = 56 bytes of model storage;
//   four cstr payloads 12+9+9+6 = 36 bytes (with terminators);
//   report.lf scalars live in frames or scratch per the backend's report.
//
// Spec notes:
// - carried is module-level because locals cannot own aggregates (§11.4);
//   the frame report shows what that decision saved and spent.
// - sortByTemperature's inner while writes slot, the outer for never
//   writes pass (§10.1); each shift is an aggregate copy with snapshot
//   semantics (§7).
// - The scalar API (dayAt/temperatureAt) keeps index-based identity at
//   the border — ch 13's doctrine at module scale.
// - The print loop calls externs with `index` as its control variable.
//   This is valid only because the target profile supplies effect
//   contracts for the ROM services (§12.4); with the conservative
//   fallback, §13.2 would treat each call as a possible write to a
//   "visible" control variable (E-CONTROL-003). Whether a routine-local
//   control variable is ever "visible" to a native boundary is gate Q6.
