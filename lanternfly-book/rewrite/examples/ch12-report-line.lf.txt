// Chapter 12 companion — Report line
// Fixed labels as cstr views; a composed line as bytes you own.
// Fixture family: Static text (conformance §5.10), minus the print call.
// Verified against specification 0.4: §2.4, §3.2, §8.5, §10.1.

// Immutable words: views of NUL-terminated static bytes.
const title as near cstr = "LANTERNFLY"
const january as near cstr = "JAN"
const february as near cstr = "FEB"

const lineCapacity as u8 = 16

var line as u8[lineCapacity]
var lineLength as u8 = 0
var overflowed as boolean = false

var titleLength as u16 = 0
var febFirst as boolean = false

// Chapter 11's occupancy pattern, elements shrunk to bytes.
sub appendByte(value as u8)
    if lineLength >= lineCapacity then
        overflowed = true
        return
    end

    line[lineLength] = value
    lineLength = lineLength + 1
end

// Digits high to low: find the leading power of ten, then divide it down.
sub appendNumber(value as u16)
    var divisor as u16 = 1
    var remaining as u16 = value
    var digit as u16 = 0

    while remaining >= 10
        remaining = remaining / 10
        divisor = divisor * 10
    end

    remaining = value

    while divisor > 0
        digit = remaining / divisor
        appendByte(u8('0' + digit))
        remaining = remaining mod divisor
        divisor = divisor / 10
    end
end

// Sign as a byte, then the unsigned path: ch 3's abs closes its arc.
sub appendSigned(value as i16)
    if value < 0 then
        appendByte(u8('-'))
    end

    appendNumber(abs(value))
end

sub main()
    titleLength = length(title)      // folds to 10 at compile time
    febFirst = february < january    // byte 0: 70 < 74 -> true

    appendByte(u8('D'))
    appendByte(u8('A'))
    appendByte(u8('Y'))
    appendByte(u8(' '))
    appendNumber(12)
    appendByte(u8(':'))
    appendByte(u8(' '))
    appendSigned(-3)
end

// Trace — the composed line, byte by byte:
//   'D' 68, 'A' 65, 'Y' 89, ' ' 32, '1' 49, '2' 50, ':' 58, ' ' 32,
//   '-' 45, '3' 51
// appendNumber(12): 12 >= 10 -> divisor 10; digits 12/10 = 1 ('1'),
//   12 mod 10 = 2, 2/1 = 2 ('2').  appendNumber(0) would emit one '0'.
// Final storage: lineLength 10, overflowed false, titleLength 10,
//   febFirst true. "DAY 12: -3" exists as ten bytes the program owns.
//
// Spec notes:
// - "JAN" occupies four static bytes (74 65 78 0); the const holds the
//   view, not a copy (§3.2).
// - cstr allows comparison, length, assignment of the view — no indexing,
//   no mutation. Copying a label's bytes into `line` is therefore not
//   expressible in source (gate Q4a); a print service takes the cstr
//   directly instead (ch 15).
// - '0' + digit: character literal is an exact byte value adopting u16
//   from `digit`; u8(...) writes the acknowledged narrowing (§2.4, §3.1).
// - A service needing terminated bytes gets one explicitly:
//   appendByte(u8('\0')).
