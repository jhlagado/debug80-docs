// Chapter 2 companion — Representations
// One measured value, stored under different agreements about bits.
// Verified against specification 0.4: §2.4, §3, §3.1, §4.1, §8.1.

// One value, three spellings — all exactly 42.
const answerDecimal as u8 = 42
const answerHex as u8 = $2a
const answerBinary as u8 = %00101010

// A reading no unsigned type can hold.
const coldestReading as i16 = -40

var exact as i16 = 300
var clipped as u8 = 0
var widened as i32 = 0
var negativeOne as i8 = -1
var sameBits as u8 = 0
var calibrated as boolean = true

sub main()
    clipped = u8(exact)        // low eight bits of 300 -> 44
    widened = exact            // i16 -> i32 widens silently, sign-extends
    sameBits = u8(negativeOne) // same width, new agreement: %11111111 -> 255
end

// Trace:
//   300 = %0000000100101100 (i16)
//   clipped  = %00101100 = 44          (narrowing keeps low bits; §3.1)
//   widened  = 300                     (silent value-preserving widening; §8.1)
//   negativeOne = %11111111 (i8, -1)
//   sameBits = 255                     (same-width signedness keeps the bits)
// Final storage: clipped 44, widened 300, sameBits 255, calibrated true.
//
// Spec notes:
// - u8(exact) is the written acknowledgement of information loss (§3.1).
// - An unconverted `clipped = exact` would warn W-CONVERT-001 (§8.1).
// - `-1` under an expected i8 is range-checked as one negative value (§3.1).
// - boolean stores canonical 0/1 and has no integer conversions (§3).
