// Chapter 9 companion — Insertion sort
// The prefix is sorted; each pass grows it by one. Argued, then written.
// Verified against specification 0.4: §6, §8.4, §10.1-10.2.

const readingCount as u8 = 8

var readings as i16[readingCount] = [12, 15, 9, -3, 0, 21, 18, 7]

sub sortReadings()
    var pass as u8 = 0
    var slot as u8 = 0
    var value as i16 = 0

    // Outer contract: pass takes each unsorted element in turn —
    // half-open, so the count appears directly (ch 7's idiom).
    for pass = 1 until count(readings)
        value = readings[pass]
        slot = pass

        // Inner shifting: uncounted, data-dependent — a while by rights.
        // The left test guards both the u8 underflow and the bounds check.
        while slot > 0 and readings[slot - 1] > value
            readings[slot] = readings[slot - 1]
            slot = slot - 1
        end

        readings[slot] = value
    end
end

sub main()
    sortReadings()
end

// Invariant trace (| separates sorted prefix from untouched suffix):
//   start   [12 | 15, 9, -3, 0, 21, 18, 7]
//   pass 1  [12, 15 | 9, -3, 0, 21, 18, 7]   value 15, 0 shifts
//   pass 2  [9, 12, 15 | -3, 0, 21, 18, 7]   value 9,  2 shifts
//   pass 3  [-3, 9, 12, 15 | 0, 21, 18, 7]   value -3, 3 shifts
//   pass 4  [-3, 0, 9, 12, 15 | 21, 18, 7]   value 0,  3 shifts
//   pass 5  [-3, 0, 9, 12, 15, 21 | 18, 7]   value 21, 0 shifts
//   pass 6  [-3, 0, 9, 12, 15, 18, 21 | 7]   value 18, 1 shift
//   pass 7  [-3, 0, 7, 9, 12, 15, 18, 21]    value 7,  5 shifts
// Final storage: sorted ascending; 14 shifts in all.
//
// Spec notes:
// - Inner loop writes `slot`, so it must be a while; the outer for's
//   control variable `pass` is never written in its body (§10.1). A
//   `for each` cannot sort: shifting needs positions, not elements.
// - `slot - 1` as an index: u8 - u8 gives i16 (§3.1); the `slot > 0`
//   guard keeps it non-negative before it is ever used (§8.4 short-circuit),
//   and in range for the bounds rule (§6).
// - Each shift is one two-byte element copy; the pass table doubles as
//   the cost table.
