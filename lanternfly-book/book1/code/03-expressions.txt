// expressions.lafy — Book 1, Chapter 3 companion listing.

var previousReading as u8 = 250
var currentReading as u8 = 20
var changeMagnitude as u16 = 0
var wideValue as i16 = 300
var byteValue as u8 = 0
var area as u16 = 1600
var side as u16 = 0

sub measureChange()
    changeMagnitude = abs(currentReading - previousReading)
end

sub convertWideValue()
    byteValue = u8(wideValue)
end

sub measureSquare()
    side = sqrt(area)
end

sub main()
    measureChange()
    convertWideValue()
    measureSquare()
end
