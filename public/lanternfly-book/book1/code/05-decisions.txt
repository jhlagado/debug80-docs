enum Status as u8
    working
    complete
    failed
end

enum ReportMode as u8
    compact
    detailed
    diagnostic
end

range VerboseMode as ReportMode = detailed to diagnostic

var itemsRemaining as u16 = 40
var errorCount as u8 = 0
var status as Status = working
var reportMode as ReportMode = compact
var lineWidth as u8 = 40
var includeChecks as boolean = false

sub updateStatus()
    if errorCount > 0 then
        status = failed
    else if itemsRemaining = 0 then
        status = complete
    else
        status = working
    end
end

// Every member has a case, so this select is complete without an else.
sub configureReport()
    select reportMode
    case compact
        lineWidth = 40
        includeChecks = false
    case detailed
        lineWidth = 80
        includeChecks = false
    case diagnostic
        lineWidth = 80
        includeChecks = true
    end
end

// A code read from a device stays u8; else handles unmatched values.
var channelCode as u8 = 0
var buffered as boolean = false

sub configureChannel()
    select channelCode
    case 0, 1
        buffered = true
    case 2 to 9
        buffered = false
    else
        buffered = true
    end
end

sub main()
    updateStatus()
    configureReport()
    configureChannel()
end
