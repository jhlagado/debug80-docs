var total as i16 = 0
var leftValue as u16 = 84
var rightValue as u16 = 30
var greatestCommonDivisor as u16 = 0
var startValue as i16 = 13
var result as i16 = 0

sub sumTen()
    var number as i16

    total = 0
    for number = 1 to 10
        total = total + number
    end
end

sub calculateGcd()
    var remainder as u16

    while rightValue <> 0
        remainder = leftValue mod rightValue
        leftValue = rightValue
        rightValue = remainder
    end

    greatestCommonDivisor = leftValue
end

sub findNextMultiple()
    var candidate as i16 = startValue

    while true
        candidate = candidate + 1

        if candidate mod 8 = 0 then
            result = candidate
            exit
        end
    end
end

enum Phase as u8
    warmup
    working
    cooldown
end

var phase as Phase = warmup
var phasesVisited as u8 = 0

// An enumeration control advances by ordinal position.
sub visitPhases()
    for phase = warmup to cooldown
        phasesVisited = phasesVisited + 1
    end
end


var oddTotal as i16 = 0
var found as boolean = false
var target as u8 = 6

// continue skips even numbers; the loop adds 1, 3, 5, 7 and 9.
sub sumOdds()
    var number as i16

    oddTotal = 0
    for number = 1 to 10
        if number mod 2 = 0 then
            continue
        end

        oddTotal = oddTotal + number
    end
end

// A flag carries an inner result out of a nested loop.
sub findInGrid()
    var row as u8
    var column as u8

    found = false
    for row = 0 until 3
        if found then
            continue
        end

        for column = 0 until 4
            if row * 4 + column = target then
                found = true
                exit
            end
        end
    end
end

sub main()
    sumTen()
    calculateGcd()
    findNextMultiple()
    visitPhases()
    sumOdds()
    findInGrid()
end
