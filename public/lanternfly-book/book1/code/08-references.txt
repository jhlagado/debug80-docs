record Monster
    var x as u8
    var y as u8
    var timer as u8
    var frame as u8
end

const clearPlane as u8[8] = [0, 0, 0, 0, 0, 0, 0, 0]

var monsters as Monster[4]
var selectedIndex as i16 = 0
var current as near ref Monster = ref monsters[0]

var boardRed as u8[8]
var boardGreen as u8[8]
var boardBlue as u8[8]
var boardPlanes as (near ref (u8[8]))[3] = [
    ref boardRed,
    ref boardGreen,
    ref boardBlue
]

sub selectMonster(index as i16)
    current = ref monsters[index]
    current.timer = 10
end

sub resetSelected()
    ref monster as Monster = monsters[selectedIndex]

    monster.timer = 0
    monster.frame = 0
end

sub clearPlanes()
    var planeIndex as i16

    for planeIndex = 0 to count(boardPlanes) - 1
        value(boardPlanes[planeIndex]) = clearPlane
    end
end
