record Actor
    var x as i16
    var y as i16
    var timer as u8
end

extern sub screenClear()
extern sub drawPixel(x as u8, y as u8, colour as u8)
extern sub showNumber(number as u16)

var playerX as u8 = 0
var playerY as u8 = 0
var playerColour as u8 = 7
var targetX as u8 = 0
var distance as u16 = 0
var root as u16 = 0
var area as u16 = 0
var framebuffer as u8[256]

const actorBytes as u8 = size(type Actor)
const timerOffset as u8 = offset(Actor.timer)

asm
ROM_WAIT_KEY = $0038
end

sub updateMeasurements()
    distance = abs(targetX - playerX)
    root = sqrt(area)
end

sub showPlayer()
    clear(framebuffer)
    screenClear()
    drawPixel(playerX, playerY, playerColour)
    showNumber(distance)
end

sub waitForKey()
    asm
        call ROM_WAIT_KEY
    end
end
