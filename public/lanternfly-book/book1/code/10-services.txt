record Date
    year as u16
    month as u8
    day as u8
end

record Reading
    value as i16
    unit as u8
    quality as u8
end

extern sub writeUnsigned(value as u16)
extern sub writeNewLine()

var previousReading as i16 = 250
var currentReading as i16 = 20
var change as u16 = 0
var workspace as u8[32]

const dateBytes as u8 = size(type Date)
const qualityOffset as u8 = offset(Reading.quality)

asm
ROM_WAIT_KEY = $0038
end

sub measureChange()
    change = abs(currentReading - previousReading)
end

sub showReading(value as u16)
    writeUnsigned(value)
    writeNewLine()
end

sub waitForKey()
    asm
        call ROM_WAIT_KEY
    end
end

sub main()
    clear(workspace)
    measureChange()
    showReading(change)
    waitForKey()
end
