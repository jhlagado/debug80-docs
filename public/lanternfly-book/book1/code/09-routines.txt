record Actor
    var x as i16
    var velocityX as i16
    var active as boolean
end

var score as i16 = 0
var bonus as i16 = 10
var selectedRow as u8[8]

sub addScore(amount as i16)
    score = score + amount
end

sub limited(value as i16, limit as i16) as i16
    if value > limit then
        return limit
    end

    return value
end

sub distanceSquared(x as i16, y as i16) as i32
    var xSquared as i32
    var ySquared as i32

    xSquared = i32(x) * i32(x)
    ySquared = i32(y) * i32(y)
    return xSquared + ySquared
end

sub clearRow(row as u8[8])
    var index as i16

    for index = 0 to count(row) - 1
        row[index] = 0
    end
end

sub updateActor(actor as Actor)
    if not actor.active then
        return
    end

    actor.x = actor.x + actor.velocityX
end

sub updateScore()
    addScore(bonus)
    score = limited(score, 1000)
    clearRow(selectedRow)
end
