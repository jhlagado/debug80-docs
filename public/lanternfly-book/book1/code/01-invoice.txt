// Chapter 1 companion — Invoice
// A program is stored data plus statements that change it.

// Fixed facts: prices in whole pence. Scalar constants occupy no storage.
const applePrice as u16 = 120
const breadPrice as u16 = 95
const milkPrice as u16 = 78

// Changing facts: the invoice's state. Static storage, installed before entry.
var total as u16 = 0
var itemCount as u8 = 0

// The build names this parameterless routine as the program entry.
sub main()
    total = total + applePrice
    itemCount = itemCount + 1

    total = total + breadPrice
    itemCount = itemCount + 1

    total = total + milkPrice
    itemCount = itemCount + 1
end

// Trace (final storage is the program's answer):
//   entry          total = 0    itemCount = 0
//   after apple    total = 120  itemCount = 1
//   after bread    total = 215  itemCount = 2
//   after milk     total = 293  itemCount = 3
// Returning from main invokes the target's program-termination service.
