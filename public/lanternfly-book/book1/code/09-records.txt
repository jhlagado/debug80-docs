record Date
    year as u16
    month as u8
    day as u8
end

record Reading
    value as i16
    unit as u8
    quality as u8
end

record DailyLog
    date as Date
    entries as Reading[4]
    used as u8
end

// The string field occupies its exact 14 bytes, so Station is 15.
record Station
    name as string[12]
    id as u8
end

const releaseDate as Date = Date(
    year = 2026,
    month = 7,
    day = 30
)

var reportDate as Date
var savedDate as Date
var readings as Reading[4]
var dailyLog as DailyLog
var station as Station
var entryIndex as i16 = 2

sub setReportDate()
    reportDate.year = 2026
    reportDate.month = 7
    reportDate.day = 30
end

sub prepareEntry()
    readings[entryIndex].value = -12
    readings[entryIndex].unit = 1
    readings[entryIndex].quality = 0
end

sub saveEntry()
    dailyLog.date = reportDate
    dailyLog.entries[entryIndex] = readings[entryIndex]
    dailyLog.entries[entryIndex].quality = 1
    dailyLog.used = 3
    savedDate = reportDate
end

sub labelStation()
    station.name = "NORTH"
    append(station.name, '2')
    station.id = 2
end

sub startNewLog()
    clear(dailyLog)
end

sub main()
    startNewLog()
    setReportDate()
    prepareEntry()
    saveEntry()
    reportDate.day = 31
    labelStation()
end
