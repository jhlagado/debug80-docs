record Reading
    value as i16
    quality as u8
end

const qualityInvalid as u8 = 0
const maximumReading as i16 = 1000

var total as i16 = 0
var workspace as u8[8]
var currentReading as Reading

sub addToTotal(amount as i16)
    total = total + amount
end

sub atMost(input as i16, maximum as i16) as i16
    if input > maximum then
        return maximum
    end

    return input
end

sub absoluteDifference(left as i16, right as i16) as u16
    var difference as i16 = left - right

    return abs(difference)
end

sub clearBlock(block as u8[8])
    var index as i16

    for index = 0 until count(block)
        block[index] = 0
    end
end

sub normaliseReading(reading as Reading)
    if reading.quality = qualityInvalid then
        return
    end

    reading.value = atMost(reading.value, maximumReading)
end

forward sub updateEnemies()

var playerClash as boolean = false
var enemyClash as boolean = false

sub updatePlayer()
    if playerClash then
        playerClash = false
        updateEnemies()
    end
end

sub updateEnemies()
    if enemyClash then
        enemyClash = false
        updatePlayer()
    end
end

sub main()
    addToTotal(10)
    addToTotal(25)
    total = atMost(total, 30)
    clearBlock(workspace)
    normaliseReading(currentReading)
    updatePlayer()
end
