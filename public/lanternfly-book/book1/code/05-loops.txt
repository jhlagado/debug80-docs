var total as i16 = 0
var evenCount as i16 = 0
var startValue as i16 = 13
var result as i16 = 0

sub sumTen()
    var number as i16

    total = 0
    for number = 1 to 10
        total = total + number
    end
end

sub countEvenColumns()
    var column as i16

    evenCount = 0
    for column = 0 to 14 step 2
        evenCount = evenCount + 1
    end
end

sub countDown()
    var remaining as i16 = 3

    while remaining > 0
        remaining = remaining - 1
    end
end

sub findNextMultiple()
    var candidate as i16 = startValue

    loop
        candidate = candidate + 1

        if candidate mod 8 = 0 then
            result = candidate
            exit
        end
    end
end
