const finished as u8 = 0
const playing as u8 = 1
const won as u8 = 2

const north as u8 = 0
const east as u8 = 1
const south as u8 = 2
const west as u8 = 3

var lives as u8 = 3
var score as u8 = 0
var status as u8 = playing
var direction as u8 = north
var deltaX as i8 = 0
var deltaY as i8 = 0

sub updateStatus()
    if lives = 0 then
        status = finished
    else if score >= 100 then
        status = won
    else
        status = playing
    end
end

sub findStep()
    deltaX = 0
    deltaY = 0

    select direction
    case north
        deltaY = -1
    case east
        deltaX = 1
    case south
        deltaY = 1
    case west
        deltaX = -1
    else
        deltaX = 0
        deltaY = 0
    end
end
