const statusWorking as u8 = 0
const statusComplete as u8 = 1
const statusFailed as u8 = 2

const modeCompact as u8 = 0
const modeDetailed as u8 = 1
const modeDiagnostic as u8 = 2

var itemsRemaining as u16 = 40
var errorCount as u8 = 0
var status as u8 = statusWorking
var reportMode as u8 = modeCompact
var lineWidth as u8 = 40
var includeChecks as boolean = false

sub updateStatus()
    if errorCount > 0 then
        status = statusFailed
    else if itemsRemaining = 0 then
        status = statusComplete
    else
        status = statusWorking
    end
end

sub configureReport()
    select reportMode
    case modeCompact
        lineWidth = 40
        includeChecks = false
    case modeDetailed
        lineWidth = 80
        includeChecks = false
    case modeDiagnostic
        lineWidth = 80
        includeChecks = true
    else
        lineWidth = 40
        includeChecks = true
    end
end

enum InputKind as u8
    serial
    file
    keyboard
end

var inputKind as InputKind = serial
var buffered as boolean = false

// Every member has a case, so this select is complete without an else.
sub configureInput()
    select inputKind
    case serial, file
        buffered = true
    case keyboard
        buffered = false
    end
end

sub main()
    updateStatus()
    configureReport()
    configureInput()
end
