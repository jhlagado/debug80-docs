// number-entry.lafy — Book 1, Chapter 14 companion listing.
// Number entry with an error set, fail and on error.
// Error handling is the newest part of the language and provisional.
//
// Line discipline: carriage return and line feed each end a line, and a
// line containing no digits is skipped, so targets that end lines with
// CR, LF or CRLF all behave identically within this entry protocol. A
// failing routine consumes the rest of its spoiled line before failing,
// so a retry starts clean.

import "standard/text-output.lafy"
import "standard/text-input.lafy"

const lineFeed as u8 = 10
const carriageReturn as u8 = 13

enum EntryError as u8
    badDigit
    tooLarge
end

var speed as u16 = 100

sub isLineEnd(key as u8) as boolean
    return key = lineFeed or key = carriageReturn
end

sub skipRestOfLine(key as u8)
    while not isLineEnd(key)
        key = readCharacter()
    end
end

sub readNumber() as u16 fails EntryError
    var value as u16 = 0
    var sawDigit as boolean = false
    var key as u8 = 0

    while true
        key = readCharacter()

        if isLineEnd(key) then
            if sawDigit then
                return value
            end

            continue
        end

        if key < '0' or key > '9' then
            skipRestOfLine(key)
            fail badDigit
        end

        if value > 6553 or (value = 6553 and key > '5') then
            skipRestOfLine(key)
            fail tooLarge
        end

        value = value * 10 + u16(key - '0')
        sawDigit = true
    end
end

sub main()
    while true
        writeText("SPEED? ")

        speed = readNumber()
        on error code
            select code
            case badDigit
                writeText("DIGITS 0 TO 9 ONLY")
            case tooLarge
                writeText("65535 IS THE LARGEST")
            end

            writeNewline()
            continue
        end

        writeText("SET")
        writeNewline()
        return
    end
end
