record Date
    var year as u16
    var month as u8
    var day as u8
end

record Reading
    var value as i16
    var unit as u8
    var quality as u8
end

record DailyLog
    var date as Date
    var entries as Reading[4]
    var used as u8
end

const releaseDate as Date = Date(
    year = 2026,
    month = 7,
    day = 30
)

var reportDate as Date
var savedDate as Date
var readings as Reading[4]
var dailyLog as DailyLog
var entryIndex as i16 = 2

sub saveEntry()
    dailyLog.entries[entryIndex] = readings[entryIndex]
    dailyLog.entries[entryIndex].quality = 1
    savedDate = reportDate
end

sub main()
    saveEntry()
end
