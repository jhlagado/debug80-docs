record Point
    var x as i8
    var y as i8
end

record Monster
    var x as u8
    var y as u8
    var direction as u8
    var state as u8
    var timer as u8
    var frame as u8
end

record Mover
    var position as Point
    var previous as Point[4]
    var savedCount as u8
end

var position as Point
var savedPosition as Point
var monsters as Monster[4]
var player as Mover
var saveIndex as i16 = 0

sub savePosition()
    player.previous[saveIndex] = player.position
    savedPosition = player.position
end
