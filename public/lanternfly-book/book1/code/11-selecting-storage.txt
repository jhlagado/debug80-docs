record Reading
    value as i16
    unit as u8
    quality as u8
end

const unitCelsius as u8 = 0
const calibrationOffset as i16 = 2
const bufferCount as u8 = 3
const bufferSize as u8 = 16

var readings as Reading[4]
var selectedReading as u8 = 0
var previousReading as u8 = 0
var nextCandidate as u8 = 2

var buffers as u8[bufferCount, bufferSize]

sub markSelected()
    alias reading as Reading = readings[selectedReading]

    reading.quality = 1
    reading.unit = unitCelsius
end

sub adjustSelected()
    alias reading as Reading = readings[selectedReading]

    reading.value = reading.value + calibrationOffset
    reading.quality = 1
end

sub clearBuffer(selected as u8)
    var index as u8

    for index = 0 until bufferSize
        buffers[selected, index] = 0
    end
end

sub advanceSelection()
    previousReading = selectedReading
    selectedReading = nextCandidate
end

enum ActiveLog as u8
    inputLog
    archiveLog
end

var activeLog as ActiveLog = inputLog
var inputEntries as u8 = 3
var archiveEntries as u8 = 7
var activeCount as u8 = 0

sub countActiveEntries() as u8
    select activeLog
    case inputLog
        return inputEntries
    case archiveLog
        return archiveEntries
    end
end

sub main()
    advanceSelection()
    markSelected()
    adjustSelected()
    clearBuffer(1)
    activeCount = countActiveEntries()
end
