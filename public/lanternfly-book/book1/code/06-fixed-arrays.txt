const dayCount as u8 = 7
const readingCount as u8 = 4

const daysInMonth as u8[12] = [
    31, 28, 31, 30, 31, 30,
    31, 31, 30, 31, 30, 31
]

var greeting as string[12] = "HELLO"
var playerName as string[12]
var statusLine as string[16]

var samples as u8[8]
var sampleTotal as u16 = 0
var readings as u16[5]
var weeklyReadings as u16[dayCount, readingCount]

sub prepareSamples()
    var index as i16

    for index = 0 until count(samples)
        samples[index] = u8(index * 2)
    end
end

sub sumSamples()
    sampleTotal = 0

    for each sample in samples
        sampleTotal = sampleTotal + sample
    end
end

sub prepareWeek()
    clear(weeklyReadings)
    weeklyReadings[1, 2] = 120
end

enum ReportMode as u8
    compact
    detailed
    diagnostic
end

// One line width per report mode: a table looked up by name.
var modeWidths as u8[ReportMode] = [40, 80, 80]

// Indexed naturally by day of the month: no entry zero, no subtracted one.
var octoberReadings as i16[1 to 31]

sub prepareOctober()
    var day as u8

    for day = lower(octoberReadings) to upper(octoberReadings)
        octoberReadings[day] = 0
    end
end

sub renameGuest()
    playerName = greeting
    append(playerName, '!')
end

// Builds text such as "MODE 2" in place: clear, a literal, a computed digit.
sub reportMode(mode as u8)
    clear(statusLine)
    append(statusLine, "MODE ")
    append(statusLine, '0' + mode)
end

sub main()
    prepareSamples()
    sumSamples()
    prepareWeek()
    prepareOctober()
    renameGuest()
    reportMode(2)
end
