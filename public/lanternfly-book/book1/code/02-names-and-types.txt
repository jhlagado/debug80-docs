const startingLives as u8 = 3
const maximumScore as u16 = 9999

var lives as u8 = startingLives
var score as u16 = 0
var gameOver as boolean = false

sub loseLife()
    if lives > 0 then
        lives = lives - 1
    end

    gameOver = lives = 0
end

sub addScore()
    if score < maximumScore then
        score = score + 10
    end
end
