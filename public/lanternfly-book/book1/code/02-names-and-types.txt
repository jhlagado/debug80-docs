CONST StartingLives AS BYTE = 3
CONST MaximumScore AS WORD = 9999

DIM Lives AS BYTE = StartingLives
DIM Score AS WORD = 0

SUB LoseLife()
    IF Lives > 0 THEN
        Lives = BYTE(Lives - 1)
    END IF
END SUB

SUB AddScore()
    IF Score < MaximumScore THEN
        Score = WORD(Score + 10)
    END IF
END SUB
